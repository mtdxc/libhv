find_package(PkgConfig QUIET)
# 查找 SDL3 是否安装
if(PKG_CONFIG_FOUND)
  pkg_check_modules(SDL3 QUIET IMPORTED_TARGET sdl3)
  if(SDL3_FOUND)
    set(SDL3_LIBS PkgConfig::SDL3)
    message(STATUS "found library: ${SDL3_LIBRARIES}")
  endif()
endif()

if(NOT SDL3_FOUND)
  find_package(SDL3 QUIET)
  if(SDL3_FOUND)
    set(SDL3_LIBS SDL3::SDL3)
    message(STATUS "found library: ${SDL3_LIBRARY}")
  endif()
endif()

if(NOT SDL3_FOUND)
  message(WARNING "${APP_NAME} disabled, please install sdl3 ffmpeg/libavcodec ffmpeg/libavutil ffmpeg/libswresample")
  return()
endif()

# 查找 FFmpeg 相关库
set(FFMPEG_LIBS "")
set(FFMPEG_FOUND TRUE)
set(FFMPEG_REQUIRED_MODULES libavcodec libavutil libswresample libswscale libavfilter libavformat)

if(PKG_CONFIG_FOUND)
  foreach(MOD ${FFMPEG_REQUIRED_MODULES})
    pkg_check_modules(${MOD} QUIET IMPORTED_TARGET ${MOD})
    if(NOT ${MOD}_FOUND)
      set(FFMPEG_FOUND FALSE)
      message(STATUS "FFmpeg module NOT found: ${MOD}")
    else()
      list(APPEND FFMPEG_LIBS PkgConfig::${MOD})
      message(STATUS "found library: ${${MOD}_LIBRARIES}")
    endif()
  endforeach()
else()
  # fallback: 使用 find_package
  find_package(FFmpeg QUIET)
  if(FFmpeg_FOUND)
    list(APPEND FFMPEG_LIBS ${FFMPEG_LIBRARIES})
    include_directories(SYSTEM ${FFMPEG_INCLUDE_DIRS})
  else()
    set(FFMPEG_FOUND FALSE)
  endif()
endif()

if(FFMPEG_FOUND)
  set(ENABLE_FFMPEG ON)
  add_definitions(-DENABLE_FFMPEG)
  message(STATUS "FFmpeg enabled")
else()
  set(ENABLE_FFMPEG OFF)
  message(WARNING "FFmpeg not fully found, ffmpeg-dependent targets disabled")
  return()
endif()

include_directories("..")
set(SRC_LIST "SDLCapture.cpp" "SDLCapture.h" "AvCapture.h" "Transcode.cpp" "Transcode.h")
#aux_source_directory(. SRC_LIST)
add_executable(uvc_server ${SRC_LIST} uvc_server.cpp)
target_link_libraries(uvc_server PRIVATE ${SDL3_LIBS} ${FFMPEG_LIBS} ice)

add_executable(sdl_player ${SRC_LIST} sdl_player.cpp)
