# ===== 查找 SDL3 =====
# 优先使用 find_path/find_library (支持 CMAKE_PREFIX_PATH / MEDIA_PREFIX)
find_path(SDL3_INCLUDE_DIR NAMES SDL3/SDL.h)
find_library(SDL3_LIBRARY NAMES SDL3)

if(SDL3_INCLUDE_DIR AND SDL3_LIBRARY)
  set(SDL3_FOUND TRUE)
  set(SDL3_LIBS ${SDL3_LIBRARY})
  include_directories(${SDL3_INCLUDE_DIR})
  message(STATUS "SDL3 found via find_path: include=${SDL3_INCLUDE_DIR}, lib=${SDL3_LIBRARY}")
else()
  # fallback: pkg-config
  if(PKG_CONFIG_FOUND)
    pkg_check_modules(SDL3 QUIET IMPORTED_TARGET sdl3)
    if(SDL3_FOUND)
      set(SDL3_LIBS PkgConfig::SDL3)
      message(STATUS "SDL3 found via pkg-config: ${SDL3_LIBRARIES}")
    endif()
  endif()
endif()

if(NOT SDL3_FOUND)
  # fallback: find_package
  find_package(SDL3 QUIET)
  if(SDL3_FOUND)
    set(SDL3_LIBS SDL3::SDL3)
    message(STATUS "SDL3 found via find_package: ${SDL3_LIBRARY}")
  endif()
endif()

if(NOT SDL3_FOUND)
  message(WARNING "sdl_player disabled, please install SDL3")
  return()
endif()

# ===== 查找 FFmpeg =====
set(FFMPEG_MODULES libavcodec libavutil libswresample libswscale)
set(FFMPEG_LIBS "")
set(FFMPEG_FOUND TRUE)

# 优先使用 find_path/find_library
find_path(FFMPEG_INCLUDE_DIR NAMES libavcodec/avcodec.h)
foreach(MOD ${FFMPEG_MODULES})
  string(REGEX REPLACE "^lib" "" MOD_SHORT ${MOD})
  find_library(${MOD}_LIBRARY NAMES ${MOD_SHORT})
  if(NOT ${MOD}_LIBRARY)
    set(FFMPEG_FOUND FALSE)
    message(STATUS "FFmpeg module NOT found: ${MOD}")
  else()
    list(APPEND FFMPEG_LIBS ${${MOD}_LIBRARY})
    message(STATUS "found ${MOD}: ${${MOD}_LIBRARY}")
  endif()
endforeach()

if(FFMPEG_FOUND AND FFMPEG_INCLUDE_DIR)
  include_directories(${FFMPEG_INCLUDE_DIR})
  message(STATUS "FFmpeg found via find_path: include=${FFMPEG_INCLUDE_DIR}")
else()
  # fallback: pkg-config
  set(FFMPEG_LIBS "")
  set(FFMPEG_FOUND TRUE)
  if(PKG_CONFIG_FOUND)
    foreach(MOD ${FFMPEG_MODULES})
      pkg_check_modules(${MOD} QUIET IMPORTED_TARGET ${MOD})
      if(NOT ${MOD}_FOUND)
        set(FFMPEG_FOUND FALSE)
        message(STATUS "FFmpeg module NOT found via pkg-config: ${MOD}")
      else()
        list(APPEND FFMPEG_LIBS PkgConfig::${MOD})
        message(STATUS "found via pkg-config: ${${MOD}_LIBRARIES}")
      endif()
    endforeach()
  else()
    set(FFMPEG_FOUND FALSE)
  endif()
endif()

if(FFMPEG_FOUND)
  set(ENABLE_FFMPEG ON)
  add_definitions(-DENABLE_FFMPEG)
  message(STATUS "FFmpeg enabled")
else()
  set(ENABLE_FFMPEG OFF)
  message(WARNING "FFmpeg not fully found, ffmpeg-dependent targets disabled")
  return()
endif()
